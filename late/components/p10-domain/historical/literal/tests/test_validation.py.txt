def model_verdict(p):
 try:
  data=decode_words([p[k] for k in M['persisted_properties']])
  if encode_save(data)!=p:return 0
  P10World.load(data);return 1
 except (ValueError,TypeError,KeyError):return 0

class ValidationTests(unittest.TestCase):
 def test_compiled_validator_against_model_and_corrupt_words(self):
  valid=snapshots();corpus=[('reachable',s) for s in valid]
  seeds=[valid[0],valid[len(valid)//2],max(valid,key=lambda p:sum(bool(v) for v in p.values()))]
  for seed in seeds:
   for key in M['persisted_properties']:
    for value in (-1,0,1,2,32767):
     p=seed.copy();p[key]=value;corpus.append((key+'='+str(value),p))
  # Pairwise caption collisions, duplicate rack objects and nonprefix receipts.
  for a,b in itertools.combinations(range(4,len(M['serials'])),2):
   if (a+b)%7:continue
   p=seeds[-1].copy()
   for limb in ('Hi','Lo'):p['p10'+M['serials'][a]+limb]=p['p10'+M['serials'][b]+limb]
   corpus.append((f'collision-{a}-{b}',p))
  machine=Compiled(HERE,992,source_vm.DEFAULTS);seen=set();accepted=rejected=0
  for label,p in corpus:
   key=tuple(p.values())
   if key in seen:continue
   seen.add(key);expected=model_verdict(p);allprops=source_vm.DEFAULTS.copy();allprops.update(p)
   actual=machine.execute('p10Validate',allprops)
   if actual!=expected:
    machine.trace=[];machine.execute('p10Validate',allprops)
    (HERE/'evidence/validation-failure.json').write_text(json.dumps(dict(label=label,expected=expected,actual=actual,properties=p,trace=machine.trace),indent=2))
   self.assertEqual(actual,expected,(label,p))
   accepted+=expected;rejected+=not expected
  for p in valid:
   v=source_vm.VM();v.p.update(p);self.assertEqual(v.call('p10Validate'),1)
  report=dict(scope='Actual emitted SCI1.1 validator versus accepted Python save model; no native playability claim',reachable_states=len(valid),unique_checks=len(seen),accepted=accepted,rejected=rejected,strict_stack_balance=True,executed_instructions=machine.steps,compiled_sha256=hashlib.sha256(machine.blob).hexdigest(),mismatches=0)
  (HERE/'evidence/compiled-validation.json').write_text(json.dumps(report,indent=2)+'\n')
